    4. ViewModel (ui/CosaViewModel.kt)

    ui/COSAViewModel.kt
    package com.example.MIAPP.ui

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue
import androidx.lifecycle.ViewModel
import com.example.MIAPP.modelo.COSA
import com.example.MIAPP.modelo.COSAUIState
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

class COSAViewModel : ViewModel() {

    // 1) Estado: privado y modificable dentro / público y de solo lectura fuera
    private val _uiState = MutableStateFlow(COSAUIState())
    val uiState: StateFlow<COSAUIState> = _uiState.asStateFlow()

    private var siguienteId = 1

    // 2) Lo que el usuario está escribiendo en el formulario
    var nombre by mutableStateOf("")
        private set
    var cantidad by mutableStateOf("")      // en el TextField siempre String
        private set

    // 3) Funciones que llama la UI cuando el usuario hace algo
    fun actualizarNombre(valor: String) { nombre = valor }
    fun actualizarCantidad(valor: String) { cantidad = valor }

    fun anadir() {
        if (nombre.isBlank()) {
            _uiState.update { it.copy(error = "El nombre es obligatorio") }
            return
        }
        val nueva = COSA(
            id = siguienteId++,
            nombre = nombre.trim(),
            cantidad = cantidad.toIntOrNull() ?: 0
        )
        // ¡Siempre una lista NUEVA! (lista + elemento)
        _uiState.update { it.copy(lista = it.lista + nueva, error = "") }
        nombre = ""
        cantidad = ""
    }

    fun borrar(cosa: COSA) {
        _uiState.update { estado -> estado.copy(lista = estado.lista.filter { it.id != cosa.id }) }
    }

    fun reiniciar() {
        _uiState.value = COSAUIState()
    }
}

    Estado en Compose: remember, mutableStateOf, recomposición

    • Compose redibuja (recomposición) una función cuando cambia un estado que lee.
    • mutableStateOf(valor) crea un estado observable. remember { } hace que se conserve entre recomposiciones.
    • by permite usarlo como variable normal (necesita imports getValue y setValue).
    • remember se pierde al girar el móvil. Para que no se pierda: rememberSaveable o, mejor, ViewModel (UD4).
    • State hoisting ("elevar el estado"): el hijo no guarda el estado; recibe el valor y una lambda para cambiarlo (value + onValueChange). Así funciona CampoEdicionPrecio del ejercicio IVA.

    Arquitectura recomendada

    • Separación de responsabilidades: cada clase una función. La UI se controla a partir de un modelo.
    • Capa de UI: elementos (funciones @Composable) + contenedores de estado (ViewModel).
    • Capa de datos: guarda, recupera y expone datos (en el curso: datos/Datos.kt). Opcional: capa de dominio.
    • Flujo unidireccional: el ViewModel expone el estado → la UI lo dibuja → la UI notifica eventos al ViewModel → el ViewModel actualiza el estado → la UI se redibuja.
    estado (uiState) ┌────────────┐ ──────────────────────► ┌────────────────────┐ │ ViewModel │ │ UI (@Composable) │ │ + UIState │ ◄────────────────────── │ Text, Button... │ └────────────┘ eventos (onClick → └────────────────────┘ viewModel.anadir())

    ViewModel + UIState + StateFlow, paso a paso

    1. UIState (modelo/XUIState.kt): data class con TODO lo que pinta la pantalla, con valores por defecto.
    2. ViewModel (ui/XViewModel.kt): class XViewModel : ViewModel().
    3. Dentro: private val _uiState = MutableStateFlow(XUIState()) → solo el ViewModel lo modifica.
    4. val uiState: StateFlow<XUIState> = _uiState.asStateFlow() → la UI lo lee (solo lectura).
    5. Para cambiarlo: _uiState.update { estadoActual -> estadoActual.copy(campo = nuevoValor) }. copy crea un objeto nuevo cambiando solo ese campo. (También _uiState.value = XUIState() para reiniciar.)
    6. Lo que el usuario escribe en un TextField se suele guardar en el ViewModel con var titulo by mutableStateOf("") + private set + función actualizarTitulo().
    7. En la UI: parámetro miViewModel: XViewModel = viewModel() y val uiState by miViewModel.uiState.collectAsState().
    8. La UI lee uiState.campo y llama a funciones del ViewModel en los onClick/onValueChange.
    ¿Por qué? Porque el ViewModel sobrevive a la rotación (cambios de configuración) y separa lógica de interfaz. "StateFlow es un flujo observable contenedor de datos que emite el estado actual y los nuevos; su propiedad value refleja el estado actual."

import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

class OperacionViewModel : ViewModel() {
    private val _operacionUItate = MutableStateFlow(OperacionUIState())

    val operacionUIState: StateFlow<OperacionUIState> = _operacionUItate.asStateFlow()

    var respuestaNumero1 by mutableStateOf("")
        private set

    var respuestaNumero2 by mutableStateOf("")
        private set

    fun actualizarNumero1(num: String) {
        respuestaNumero1 = num

        _operacionUItate.update { estadoActual ->
            estadoActual.copy(
                numero1 = respuestaNumero1
            )
        }

        calcular()

    }

    fun actualizarNumero2(num: String) {
        respuestaNumero2 = num

        _operacionUItate.update { estadoActual ->
            estadoActual.copy (
                numero2 = respuestaNumero2
            )
        }

        calcular()
    }

    private fun calcular() {
        _operacionUItate.update { estadoActual ->
            val num1 = estadoActual.numero1.toIntOrNull() ?: 0
            val num2 = estadoActual.numero2.toIntOrNull() ?: 0

            estadoActual.copy(
                resultado = (num1 + num2).toString()
            )
        }
    }
}

class JuegoViewModel : ViewModel(){

    private val _uiState = MutableStateFlow(JuegoUIState())

    val uiState: StateFlow<JuegoUIState> = _uiState.asStateFlow()

    private lateinit var palabraActual: String
    private lateinit var palabraOculta: String
    private var palabrasUsadas: MutableSet<String> = mutableSetOf()

    var letraUsuario by mutableStateOf("")
        private set

    init {
        reiniciarJuego()
    }

    fun reiniciarJuego() {
        palabrasUsadas.clear()
        _uiState.value = JuegoUIState(
            palabraActual = elegirPalabraAleatoria()
        )
    }

    private fun elegirPalabraAleatoria(): String {
        palabraActual = listaPalabras.random()

        if(palabrasUsadas.contains(palabraActual)) {
            return elegirPalabraAleatoria()
        }
        else {
            palabrasUsadas.add(palabraActual)

            palabraOculta = "_".repeat(palabraActual.length)

            return palabraOculta
        }
    }

    fun actualizarLetraUsuario(letraRespuesta: String) {
        letraUsuario = letraRespuesta.uppercase()
    }

    fun comprobarLetraUsuario() {
        if(palabraActual.contains(letraUsuario[0], ignoreCase = true)){

            val guiones = palabraOculta.toCharArray()

            // Reemplaza las ocurrencias de la letra en la posición correspondiente
            for (i in palabraActual.indices) {
                if (palabraActual[i] == letraUsuario[0]) {
                    guiones[i] = letraUsuario[0]
                }
            }
            palabraOculta = String(guiones)

            palabraActual = palabraActual.replace(letraUsuario[0], '_')

            _uiState.update { estadoActual ->
                estadoActual.copy(
                    palabraActual = palabraOculta,
                    isLetraIncorrecta = false)
            }

            if(!palabraOculta.contains('_')) {
                val puntuacionActual = _uiState.value.puntos.plus(PUNTOS_POR_ACIERTO)

                actualizarEstadoJuego(puntuacionActual)
            }
        }
        else {
            _uiState.update { estadoActual ->
                estadoActual.copy(isLetraIncorrecta = true)
            }
        }

        actualizarLetraUsuario("")
    }

    fun actualizarEstadoJuego(puntuacionActual: Int) {
        if(palabrasUsadas.size == RONDAS){
            _uiState.update { estadoActual->
                estadoActual.copy(
                    isLetraIncorrecta = false,
                    puntos = puntuacionActual,
                    isFinJuego = true
                )
            }
        }
        else {
            _uiState.update { estadoactual ->
                estadoactual.copy(
                    palabraActual = elegirPalabraAleatoria(),
                    isLetraIncorrecta = false,
                    puntos = puntuacionActual,
                    ronda = estadoactual.ronda.inc()
                )
            }
        }
    }
}

    private var siguienteId = 1

    // Campos del formulario
    var nombre by mutableStateOf("")
        private set
    var nota by mutableStateOf("")
        private set
    var modulo by mutableStateOf(listaModulos[0])
        private set
    var repetidor by mutableStateOf(false)
        private set

    fun actualizarNombre(valor: String) { nombre = valor }
    fun actualizarNota(valor: String) { nota = valor }
    fun actualizarModulo(valor: String) { modulo = valor }
    fun actualizarRepetidor(valor: Boolean) { repetidor = valor }

    fun anadirAlumno() {
        // Admitimos coma o punto como separador decimal
        val notaNumero = nota.trim().replace(',', '.').toDoubleOrNull()

        val error = when {
            nombre.isBlank() -> "El nombre es obligatorio"
            notaNumero == null -> "La nota debe ser un número"
            notaNumero < 0 || notaNumero > 10 -> "La nota debe estar entre 0 y 10"
            else -> ""
        }
        if (error.isNotEmpty() || notaNumero == null) {
            _uiState.update { it.copy(error = error) }
            return
        }

        val alumno = Alumno(siguienteId++, nombre.trim(), notaNumero, modulo, repetidor)
        _uiState.update { it.copy(alumnos = it.alumnos + alumno, error = "") }
        limpiarFormulario()
    }

    fun pedirBorrar(alumno: Alumno) {
        _uiState.update { it.copy(alumnoABorrar = alumno) }

        _uiState.update { it.copy(alumnoABorrar = null) }
    }

        _uiState.update { estado ->
            estado.copy(
                alumnos = estado.alumnos.filter { it.id != estado.alumnoABorrar?.id },
                alumnoABorrar = null
            )
        }
    }

    private fun limpiarFormulario() {
        nombre = ""
        nota = ""
        modulo = listaModulos[0]
        repetidor = false

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

class QuizViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(QuizUIState())
    val uiState: StateFlow<QuizUIState> = _uiState.asStateFlow()

    fun seleccionar(opcion: Int) {
        if (!_uiState.value.comprobada) {
            _uiState.update { it.copy(seleccionada = opcion) }
        }
    }

    fun comprobar() {
        val estado = _uiState.value
        if (estado.seleccionada == -1 || estado.comprobada) return
        val acierto = estado.seleccionada == listaPreguntas[estado.indice].correcta
        _uiState.update {
            it.copy(
                comprobada = true,
                acertada = acierto,
                puntos = if (acierto) it.puntos + PUNTOS_ACIERTO else it.puntos
            )
        }
    }

    fun siguiente() {
        _uiState.update { estado ->
            if (estado.indice + 1 >= listaPreguntas.size) {
                estado.copy(finJuego = true)
            } else {
                estado.copy(
                    indice = estado.indice + 1,
                    seleccionada = -1,
                    comprobada = false,
                    acertada = false
                )
            }
        }
    }

    fun reiniciar() {
        _uiState.value = QuizUIState()
    }
}

    • Arquitectura en capas: Aplicaciones → Framework → Sistema operativo Android (Linux) → Hardware.

    Jetpack Compose

    • Kit moderno y declarativo para crear interfaces nativas en Kotlin (sustituye a los XML de layouts).
    • Funciones de componibilidad: @Composable, no devuelven nada, nombre en mayúscula y sustantivo.
    • Recomposición: Compose vuelve a ejecutar las funciones cuyos estados han cambiado.
    • Modificadores: decoran/modifican un elemento (tamaño, padding, clic…). El orden importa.
    • sp para texto (escala con la configuración de accesibilidad), dp para tamaños.

    Arquitectura (UD4)

    • Principios: separación de problemas y controlar la IU a partir de un modelo.
    • Capas: UI (elementos + contenedores de estado = ViewModel), datos, opcional dominio.
    • ViewModel: contiene y expone el estado; sobrevive a los cambios de configuración (girar).
    • StateFlow: flujo observable que emite el estado actual y los nuevos. MutableStateFlow privado, asStateFlow() lo hace de solo lectura. update lo actualiza y copy copia cambiando campos.
    • Flujo unidireccional de datos: estado baja (VM → UI), eventos suben (UI → VM).
